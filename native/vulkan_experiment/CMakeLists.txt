cmake_minimum_required(VERSION 3.24)
project(hulk_vulkan_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(ENABLE_HLSL ON CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glslang GIT_REPOSITORY https://github.com/KhronosGroup/glslang.git GIT_TAG 16.0.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(glslang)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(ROOT "../.." ABSOLUTE)
set(SCDA "${ROOT}/third_party/scda-vulkan")
set(HULK "${ROOT}/third_party/xboxxberecomp")
# Separate executables: the older reference and the newer Hulk state ABI
# must never be mixed or linked together.
add_executable(scda_renderer_reference "${SCDA}/android/tests/renderer-test.cpp" "${SCDA}/src/gpu/shaders.cpp")
target_include_directories(scda_renderer_reference PRIVATE "${SCDA}/src")
target_link_libraries(scda_renderer_reference PRIVATE Vulkan::Headers Threads::Threads glslang glslang-default-resource-limits)
target_compile_definitions(scda_renderer_reference PRIVATE ENABLE_HLSL)
target_compile_options(scda_renderer_reference PRIVATE -UNDEBUG)
add_executable(hulk_shader_probe test_shaders.cpp "${HULK}/gpu/shaders.cpp")
target_include_directories(hulk_shader_probe PRIVATE "${HULK}" "${SCDA}/src")
target_compile_definitions(hulk_shader_probe PRIVATE ENABLE_HLSL)
target_compile_options(hulk_shader_probe PRIVATE -UNDEBUG)
target_link_libraries(hulk_shader_probe PRIVATE glslang glslang-default-resource-limits)
enable_testing()
add_test(NAME reference COMMAND scda_renderer_reference)
add_test(NAME hulk_shader_compile COMMAND hulk_shader_probe "${CMAKE_CURRENT_BINARY_DIR}/spv")

add_executable(hulk_vulkan_offscreen test_offscreen.cpp)
target_include_directories(hulk_vulkan_offscreen PRIVATE "${HULK}" "${SCDA}/src")
target_compile_options(hulk_vulkan_offscreen PRIVATE -UNDEBUG)
target_link_libraries(hulk_vulkan_offscreen PRIVATE Vulkan::Vulkan Threads::Threads)
add_test(NAME hulk_vulkan_pixels COMMAND hulk_vulkan_offscreen "${CMAKE_CURRENT_BINARY_DIR}/spv")
set_tests_properties(hulk_shader_compile PROPERTIES FIXTURES_SETUP shader_files)
set_tests_properties(hulk_vulkan_pixels PROPERTIES FIXTURES_REQUIRED shader_files TIMEOUT 60)
